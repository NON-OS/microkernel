# Every gate between a guest and what is outside it, in src/linux, held by
# scripts/check_linux_gates.py. One per line, fields split by " | ":
#   gate FILE | NEEDLE | what it refuses
#   first FILE | EARLIER | LATER | why the order matters
#   reply FILE                 (a file allowed to answer a guest's frame)
# The declared calls: one outside them is EPERM, before anything else.
first serve/route_life.rs | app::outside(guest, frame.nr) | first(guest, frame) | the set before any answer
first serve/family_answer.rs | app::undeclared( | frame_in(&self.ns, frame) | the set before a pid's ESRCH
gate app/calls.rs | Some(errno::fail(errno::EPERM)) | a call outside the set is EPERM (side 1)
gate serve/unserved.rs | errno::fail(errno::ENOSYS) | a call not served is ENOSYS (side 1)
# The box (nonos_linux_policy::mask), each where its service is reached.
gate net/socket.rs | !crate::linux::app::inet(guest) | AF_INET without NET_ANON or NET_LISTEN_LO
gate net/connect/out.rs | crate::linux::app::may_dial(guest, target, to.port) | a stream out, NET_ANON and its dests
gate net/bind.rs | mask::NET_LISTEN_LO | bind without NET_LISTEN_LO
gate net/listen.rs | mask::NET_LISTEN_LO | listen without NET_LISTEN_LO
gate net/named/connect.rs | app::granted(guest, WAYLAND) | the display without WAYLAND
gate unix/sock.rs | mask::WAYLAND | the display without WAYLAND
gate call/spawn/exec.rs | mask::EXEC | execve without EXEC
gate call/mem/map_anon.rs | granted(guest, JIT) | executable anonymous memory without JIT
gate call/mem/prot.rs | mask::JIT | mprotect +x without JIT
gate call/mem/prot.rs | guest.span_unproven(start, span) | +x on file bytes never proved
gate call/mem/map_file.rs | proven(guest, req.fd) | an executable file mapping never proved
gate call/mem/prot_args.rs | if wx_refused(prot) | a page writable and executable
gate call/mem/map_args.rs | wx_refused(prot) | a mapping writable and executable
gate file/models/open.rs | mask::MODELS_RO | a model without MODELS_RO
gate file/models/stat.rs | crate::linux::app::models() | /models without MODELS_RO
gate file/private/names.rs | crate::linux::app::home() | a private home without FILES_HOME
gate file/private/rule.rs | (!HOMES.contains(p) || home) | /home and /root private only with FILES_HOME
gate start_guest.rs | app::grant(guest.app.mask) | the family's tree, from the program's mask
gate file/root.rs | !super::private::shared_writes_allowed() | the shared tree is read-only to a guest
gate call/io.rs | Some(Kind::Stdin) if !crate::linux::app::granted(guest, TTY) | typed input without TTY
gate call/ioctl.rs | mask::TTY | terminal control without TTY
gate net/socket_kind.rs | (SOCK_RAW, Domain::Inet) | raw internet sockets, always EPERM
gate net/dgram.rs | refuse_out("sendto", a) | a datagram out of the family
gate net/connect/dgram.rs | refuse_out("connect", to) | a datagram peer out of the family
gate net/msg/send.rs | refuse_out("sendmsg", *a) | a datagram out of the family
# Wayland popups (C8v): composed into their own window, grabbed only from
# the latest press, and ended by everything that ends a grab today. A
# focus-out from the input router will call end_all too (I-C8).
first wayland/popup/grab.rs | may_grab(id, serial, root) | super::layer::lay( | the grab's rules before the popup shows
gate wayland/input/button.rs | guest.scene.popups.last_press = Some(serial) | a grab naming a button press not the latest
gate wayland/input/key.rs | guest.scene.popups.last_press = Some(serial) | a grab naming a key press not the latest
gate wayland/popup/ends.rs | guest.scene.popups.last_press = None | a grab after an end, with no new press
gate wayland/popup/ends.rs | keyboard_on(guest, keyboard) | the keyboard left on an ended popup
gate wayland/commit.rs | popup::gone::unheld_commit(guest, id) | a popup shown with no grab
gate wayland/forget.rs | popup::gone::parent_gone(guest, what, id) | a grab outliving its parent
gate wayland/route/rest.rs | TOPLEVEL_SET_MINIMIZED) => popup::ends::end_all(guest) | a grab outliving a minimize
gate wayland/toplevel/ask.rs | popup::ends::end_all(guest) | a grab outliving a full-screen or maximize change
gate wayland/toplevel/destroy.rs | popup::ends::end_all(guest) | a grab outliving its toplevel
gate wayland/present/show.rs | popup::ends::end_all(guest) | a grab outliving a reshape

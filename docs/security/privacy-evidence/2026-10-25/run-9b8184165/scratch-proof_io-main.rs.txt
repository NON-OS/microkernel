//! SCRATCH ONLY, for S-13's boot: capsule_proof_io (IPC and Memory, no
//! launcher) asks for what only a launcher may have. Every refusal is the
//! kernel's to print ([CAP-DENY]); the exit code says which step went wrong.

#![no_std]
#![no_main]

use nonos_libc::{mk_exit, mk_idle_ms, mk_ipc_call_timeout, mk_service_lookup, mk_tool_run};

const EPERM: i64 = -1;

fn installer() -> Option<u64> {
    for _ in 0..1200 {
        let (mut port, mut pid) = (0u32, 0u32);
        let rc = mk_service_lookup(b"installer".as_ptr(), 9, &mut port, &mut pid);
        if rc >= 0 && port != 0 {
            return Some(port as u64);
        }
        mk_idle_ms(500);
    }
    None
}

/// One load-by-name for `name` with `caps`: the reply's status.
fn load(port: u64, caps: u64, name: &[u8]) -> i64 {
    let mut tx = [0u8; 64];
    tx[0..4].copy_from_slice(&7u32.to_le_bytes());
    tx[4..6].copy_from_slice(&4u16.to_le_bytes());
    tx[8..16].copy_from_slice(&caps.to_le_bytes());
    tx[16] = name.len() as u8;
    tx[17..17 + name.len()].copy_from_slice(name);
    let mut rx = [0u8; 32];
    let rc = mk_ipc_call_timeout(port, tx.as_ptr(), 17 + name.len(), rx.as_mut_ptr(), 32, 60_000);
    if rc < 8 {
        return -1000 + rc;
    }
    i32::from_le_bytes([rx[4], rx[5], rx[6], rx[7]]) as i64
}

#[no_mangle]
pub unsafe extern "C" fn _start() -> ! {
    // Let the desktop come up first, so the lines land after it.
    mk_idle_ms(20_000);
    if mk_tool_run(b"tool.install", b"") != EPERM {
        mk_exit(11);
    }
    if mk_tool_run(b"tool.tokei", b"") != EPERM {
        mk_exit(12);
    }
    let Some(port) = installer() else { mk_exit(13) };
    if load(port, 0, b"std_proof") != EPERM {
        mk_exit(14);
    }
    if load(port, u64::MAX, b"std_proof") != EPERM {
        mk_exit(15);
    }
    // A step past the last: a nonzero code says the proof ran to its end.
    mk_exit(42)
}
